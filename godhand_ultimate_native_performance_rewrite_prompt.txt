```cpp
            uint32x4_t sel = {0xFFFFFFFFu, 0, 0, 0};
            return vreinterpretq_m128_f32(vbslq_f32(sel, vreinterpretq_f32_m128(val), vreinterpretq_f32_m128(dst)));
        }
        case 0x4: { // Y only
            uint32x4_t sel = {0, 0xFFFFFFFFu, 0, 0};
            return vreinterpretq_m128_f32(vbslq_f32(sel, vreinterpretq_f32_m128(val), vreinterpretq_f32_m128(dst)));
        }
        case 0x2: { // Z only
            uint32x4_t sel = {0, 0, 0xFFFFFFFFu, 0};
            return vreinterpretq_m128_f32(vbslq_f32(sel, vreinterpretq_f32_m128(val), vreinterpretq_f32_m128(dst)));
        }
        case 0x1: { // W only
            uint32x4_t sel = {0, 0, 0, 0xFFFFFFFFu};
            return vreinterpretq_m128_f32(vbslq_f32(sel, vreinterpretq_f32_m128(val), vreinterpretq_f32_m128(dst)));
        }
        case 0xE: { // XYZ (most common in VU1 3D position/normal math!)
            uint32x4_t sel = {0xFFFFFFFFu, 0xFFFFFFFFu, 0xFFFFFFFFu, 0};
            return vreinterpretq_m128_f32(vbslq_f32(sel, vreinterpretq_f32_m128(val), vreinterpretq_f32_m128(dst)));
        }
        default: {
            uint32x4_t sel = {
                (mask & 8) ? 0xFFFFFFFFu : 0u,
                (mask & 4) ? 0xFFFFFFFFu : 0u,
                (mask & 2) ? 0xFFFFFFFFu : 0u,
                (mask & 1) ? 0xFFFFFFFFu : 0u
            };
            return vreinterpretq_m128_f32(vbslq_f32(sel, vreinterpretq_f32_m128(val), vreinterpretq_f32_m128(dst)));
        }
    }
}
```
  Inside `fmacF`, `fmacV`, and `fmacFastImpl`, when writing to `vf[fd]` (with `fd != 0`) or `acc`:
  ```cpp
  if (fd != 0) {
      m_state.vf[fd].v = blendMasked(m_state.vf[fd].v, res, mask);
  }
  ```
  Do NOT call `vnorm()`, `storeMaskedWithFlags()`, or `neonMovemaskPs()`! This reduces every single VU1 vector instruction from ~28 ARM64 instructions down to **2–3 ARM64 NEON instructions** (`fmul`/`fmla` + `bsl`)!

4C. Vectorize `CLIP` in `ps2_vu1_impl.inl` with 4-Lane ARM64 NEON
--------------------------------------------------------------------------------
Look at `CLIP(uint8_t fs, uint8_t ft, uint8_t fsf)` in `ps2_vu1_impl.inl`:
- Currently it extracts scalar floats `x, y, z, w = fabsf(...)` and runs 6 scalar `if` branches.
- Replace `CLIP` with branchless ARM64 NEON comparisons:
```cpp
void VU1Runtime::clip(uint8_t fs, uint8_t ft, uint8_t tf) {
    float32x4_t vfs = vreinterpretq_f32_m128(m_state.vf[fs].v);
    float w = std::fabs(vgetq_lane_f32(vreinterpretq_f32_m128(m_state.vf[ft].v), tf & 3));
    float32x4_t vw  = vdupq_n_f32(w);
    float32x4_t vnw = vdupq_n_f32(-w);
    uint32x4_t gt = vcgtq_f32(vfs, vw);   // +x > w, +y > w, +z > w
    uint32x4_t lt = vcltq_f32(vfs, vnw);  // -x < -w, -y < -w, -z < -w
    uint32_t c = (m_state.clip << 6) & 0xFFFFFFu;
    if (vgetq_lane_u32(gt, 0)) c |= 0x01u;
    if (vgetq_lane_u32(lt, 0)) c |= 0x02u;
    if (vgetq_lane_u32(gt, 1)) c |= 0x04u;
    if (vgetq_lane_u32(lt, 1)) c |= 0x08u;
    if (vgetq_lane_u32(gt, 2)) c |= 0x10u;
    if (vgetq_lane_u32(lt, 2)) c |= 0x20u;
    m_state.clip = c;
}
```

STEP 6: MOBILE TBDR VULKAN TUNING & AUTO DYNAMIC RESOLUTION IN PCSX2 GS BRIDGE
        (`Bridge.cpp`, `GSDeviceVK.cpp`, `main.cpp`)
================================================================================
Why: Mobile GPUs (Mali-G615 on Dimensity 7300, Adreno on Snapdragon) are Tile-Based Deferred Renderers (TBDR). Desktop PCSX2 settings cause expensive RenderPass splits and tile-memory flushes to DRAM whenever Date/Stencil barriers or texture readbacks happen.

1. **In `app/src/main/cpp/pcsx2_gs/Bridge.cpp` (inside `Bridge::init` / `applyOptions`)**:
   Force these mobile-TBDR-friendly `GSConfig` settings:
   ```cpp
   GSConfig.UseDebugDevice = false;
   GSConfig.VsyncQueueSize = 2;
   GSConfig.FramerateNTSC = 60.0f;
   GSConfig.GPUPaletteConversion = false;      // Huge savings on Mali/Adreno! Avoids compute/fragment CLUT pass per draw
   GSConfig.PreloadFrameWithGSData = false;    // Eliminates full-frame host->device upload each field
   GSConfig.Mipmap = HWMipmapLevel::Off;       // God Hand doesn't need HW mip generation per texture upload
   GSConfig.TrilinearFiltering = TriFiltering::Off;
   GSConfig.AccurateDATE = AccDATE::Off;       // Prevents TBDR RenderPass breaks on alpha/stencil draws!
   GSConfig.BlendingAccuracy = AccBlendLevel::Minimum; // Prevents framebuffer fetch/barrier stalls on particles & dust!
   GSConfig.HWDownloadMode = GSHardwareDownloadMode::Unsynchronized; // NEVER stall EE/GS waiting on vkQueueWaitIdle!
   GSConfig.TexturePreloading = TexturePreloadingLevel::Full;
   GSConfig.UserHacks_WildArmsOffset = false;
   GSConfig.UserHacks_SkipDraw_Start = 0;
   GSConfig.UserHacks_SkipDraw_End = 0;
   ```
   *Why `BlendingAccuracy = Minimum` and `AccurateDATE = Off` are critical for God Hand*:
   When crates break or enemies spawn in God Hand, the game draws hundreds of overlapping alpha-blended dust, spark, and aura sprites. If `BlendingAccuracy` or `AccurateDATE` is enabled, `GSDeviceVK` inserts a `vkCmdPipelineBarrier` (or splits the Vulkan RenderPass) **between every single particle sprite**, forcing Mali-G615 / Adreno to flush tile memory to system RAM hundreds of times per frame! Setting both to `Minimum` / `Off` keeps all particle draws inside a **single continuous tiled Vulkan RenderPass**!
